🎯 预测输出：

struct Point { int x{}; int y{}; };

void move(Point& p, int dx, int dy)
{
    p.x += dx;
    p.y += dy;
}

int main()
{
    Point pos{ 1, 2 };
    move(pos, 3, 4);
    std::cout << pos.x << ',' << pos.y;
}
