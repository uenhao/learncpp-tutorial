Part B · 真实调试任务（原文 Quiz Q1/Q2）

以下两个程序都要真的用调试器跑：打断点 → 单步 → 看变量 → 修复。仓库 chapter3/quiz/exercises/ 里有可直接调试的工程。

调试⑧ 程序应把两个数相加，但结果不对。用调试器找出 bug：

#include <iostream>

int readNumber(int x)
{
	std::cout << "Please enter a number: ";
	std::cin >> x;
	return x;
}

void writeAnswer(int x)
{
	std::cout << "The sum is: " << x << '\n';
}

int main()
{
	int x {};
	readNumber(x);
	x = x + readNumber(x);
	writeAnswer(x);
	return 0;
}

调试⑨ 程序应做除法。输入 8 和 4，程序崩溃。用调试器定位：

#include <iostream>

int readNumber()
{
	std::cout << "Please enter a number: ";
	int x {};
	std::cin >> x;
	return x;
}

void writeAnswer(int x)
{
	std::cout << "The quotient is: " << x << '\n';
}

int main()
{
	int x{ };
	int y{ };
	x = readNumber();
	x = readNumber();   // ← 嫌疑行
	writeAnswer(x/y);
	return 0;
}
