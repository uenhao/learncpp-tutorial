⑨ 配合重试循环（每次失败打印当前 x），用户依次输入 a、b、3，程序输出序列？

double getDouble()
{
    while (true)
    {
        std::cout << "Enter: ";
        double x{};
        std::cin >> x;
        if (!std::cin)
        {
            std::cin.clear();
            ignoreLine();
            std::cout << x << ' '; // 打印失败后的 x
            continue;
        }
        ignoreLine();
        std::cout << x;
        return x;
    }
}
// 输入: a  b  3
